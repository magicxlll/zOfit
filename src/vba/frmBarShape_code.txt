Option Explicit

Private mHandlers As Collection
Private mSelectedCode As String
Private mShapeMap As Object

Private Sub UserForm_Initialize()
    modUnicode.SetFormCaptionU Me, modUnicode.U("zOfit QS Pro - Ch\u1ecdn H\u00ecnh D\u1ea1ng C\u1ed1t Th\u00e9p Chu\u1ea9n (41 D\u00e1ng)")

    Dim ctrl As MSForms.Control
    For Each ctrl In Me.Controls
        On Error Resume Next
        ctrl.Font.Name = "Segoe UI"
        ctrl.Font.Size = 9
        On Error GoTo 0
    Next ctrl

    Me.lblGridTitle.Font.Bold = True
    Me.lblGridTitle.Caption = modUnicode.U("1. Ch\u1ecdn m\u1eabu d\u00e1ng th\u00e9p (Click \u0111\u1ec3 xem r\u00f5 L1..L7, M1, M2 | Double-click \u0111\u1ec3 th\u00eam nhanh):")
    Me.lblPrevTitle.Font.Bold = True
    Me.lblPrevTitle.Caption = modUnicode.U("2. Minh h\u1ecda chi ti\u1ebft k\u00edch th\u01b0\u1edbc \u0111o\u1ea1n th\u00e9p:")

    Me.lblSelected.Font.Bold = True
    Me.lblSelected.Font.Size = 10
    Me.lblSelected.ForeColor = RGB(0, 51, 153)
    Me.lblParams.Font.Bold = True
    Me.lblParams.ForeColor = RGB(180, 0, 0)

    Me.btnAddEnd.Caption = modUnicode.U("Th\u00eam v\u00e0o cu\u1ed1i b\u1ea3ng")
    Me.btnInsertHere.Caption = modUnicode.U("Ch\u00e8n t\u1ea1i d\u00f2ng ch\u1ecdn")
    Me.btnApplyRow.Caption = modUnicode.U("\u0110\u1ed5i d\u00e1ng d\u00f2ng \u0111ang ch\u1ecdn")
    Me.btnClose.Caption = modUnicode.U("\u0110\u00f3ng")
    Me.btnAddEnd.Font.Bold = True

    Set mShapeMap = CreateObject("Scripting.Dictionary")
    Dim tbl As Variant
    tbl = modRebarTKT.GetAllShapesTable()
    If IsArray(tbl) Then
        Dim r As Long
        For r = 1 To UBound(tbl, 1)
            mShapeMap(CStr(tbl(r, 1))) = Array(CStr(tbl(r, 2)), CStr(tbl(r, 3)), CStr(tbl(r, 4)), CStr(tbl(r, 16)))
        Next r
    End If

    ' IMPORTANT: Instantiate a NEW clsShapeBtn on every loop iteration (never Dim ... As New inside a loop!)
    Set mHandlers = New Collection
    Dim i As Long, code As String
    Dim im As MSForms.Image
    Dim h As clsShapeBtn
    For i = 1 To 41
        code = "D" & Format$(i, "00")
        Set im = Nothing
        On Error Resume Next
        Set im = Me.Controls("img" & code)
        On Error GoTo 0
        If Not im Is Nothing Then
            Set h = New clsShapeBtn
            Set h.Img = im
            Set h.ParentForm = Me
            h.ShapeCode = code
            mHandlers.Add h
        End If
    Next i

    OnShapeClicked "D01"
End Sub

Public Sub OnShapeClicked(ByVal shapeCode As String)
    mSelectedCode = shapeCode
    Dim i As Long, code As String
    Dim im As MSForms.Image
    For i = 1 To 41
        code = "D" & Format$(i, "00")
        Set im = Nothing
        On Error Resume Next
        Set im = Me.Controls("img" & code)
        On Error GoTo 0
        If Not im Is Nothing Then
            If code = shapeCode Then
                im.BorderColor = RGB(192, 0, 0)
                im.BorderStyle = fmBorderStyleSingle
                im.BackColor = RGB(255, 248, 225)
            Else
                im.BorderColor = RGB(210, 210, 210)
                im.BorderStyle = fmBorderStyleSingle
                im.BackColor = RGB(255, 255, 255)
            End If
        End If
    Next i

    ' Update large zoomed preview image
    On Error Resume Next
    Set Me.imgPreview.Picture = Me.Controls("img" & shapeCode).Picture
    On Error GoTo 0

    If mShapeMap.Exists(shapeCode) Then
        Dim info As Variant
        info = mShapeMap(shapeCode)
        Me.lblSelected.Caption = shapeCode & " - " & info(0) & vbCrLf & "(" & info(1) & ")"
        Me.lblParams.Caption = modUnicode.U("Nh\u1eadp c\u00e1c c\u1ed9t: ") & info(2)
        Me.lblFormula.Caption = modUnicode.U("C\u00f4ng th\u1ee9c chi\u1ec1u d\u00e0i: ") & Replace(info(3), "{r}", "")
    Else
        Me.lblSelected.Caption = shapeCode
        Me.lblParams.Caption = ""
        Me.lblFormula.Caption = ""
    End If
End Sub

Public Sub OnShapeDblClicked(ByVal shapeCode As String)
    OnShapeClicked shapeCode
    btnAddEnd_Click
End Sub

Private Sub btnAddEnd_Click()
    If Len(mSelectedCode) = 0 Then mSelectedCode = "D01"
    modRebarTKT.InsertBarRowFromUI mSelectedCode, False
End Sub

Private Sub btnInsertHere_Click()
    If Len(mSelectedCode) = 0 Then mSelectedCode = "D01"
    modRebarTKT.InsertBarRowFromUI mSelectedCode, True
End Sub

Private Sub btnApplyRow_Click()
    If Len(mSelectedCode) = 0 Then mSelectedCode = "D01"
    modRebarTKT.ApplyShapeToActiveRow mSelectedCode
End Sub

Private Sub btnClose_Click()
    Unload Me
End Sub
